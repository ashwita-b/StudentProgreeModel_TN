 import csv import math  class AdaptiveTest:     """Modified AdaptiveTest class for Flask integration"""          def __init__(self, csv_file, max_items=18, sem_threshold=0.7):         self.csv_file = csv_file         self.max_items = max_items         self.sem_threshold = sem_threshold                  # Test data         self.item_db = None         self.responses = []         self.difficulties = []         self.item_codes = []         self.questions_asked = []  # Store full question data         self.theta = 0.0         self.sem = 10.0          def load_items(self):         """Load questions from CSV file"""         with open(self.csv_file, newline='', encoding='utf-8') as f:             reader = csv.DictReader(f)             coditem, b, item, key = [], [], [], []             option1, option2, option3, option4 = [], [], [], []                          for row in reader:                 coditem.append(row['coditem'])                 b.append(float(row['b']))                 item.append(row['item'])                 key.append(row['key'])                 option1.append(row['option1'])                 option2.append(row['option2'])                 option3.append(row['option3'])                 option4.append(row['option4'])                  self.item_db = {             'coditem': coditem,              'b': b,              'item': item,              'key': key,             'option1': option1,             'option2': option2,             'option3': option3,             'option4': option4         }          def get_next_item(self):         """Select next item based on current ability"""         if not self.item_db['b']:             return None                  # Find closest item to current theta         differences = [abs(self.theta - b) for b in self.item_db['b']]         idx = differences.index(min(differences))                  # Extract and remove item from database         next_item = {             'coditem': self.item_db['coditem'].pop(idx),             'b': self.item_db['b'].pop(idx),             'item': self.item_db['item'].pop(idx),             'key': self.item_db['key'].pop(idx),             'option1': self.item_db['option1'].pop(idx),             'option2': self.item_db['option2'].pop(idx),             'option3': self.item_db['option3'].pop(idx),             'option4': self.item_db['option4'].pop(idx)         }         return next_item          def submit_answer(self, question_data, user_answer):         """Process user's answer and update ability estimate"""         is_correct = (user_answer == question_data['key'])         score = 1 if is_correct else 0                  # Store response data         self.responses.append(score)         self.difficulties.append(question_data['b'])         self.item_codes.append(question_data['coditem'])         self.questions_asked.append(question_data)                  # Update ability estimate         self.theta, self.sem = self.estimate_ability()                  return {             'is_correct': is_correct,             'correct_answer': question_data['key'],             'theta': self.theta,             'sem': self.sem         }          def estimate_ability(self):         """Estimate ability (theta) using Rasch model"""         if not self.responses:             return 0.0, 10.0                  conv = 0.001         J = len(self.responses)         theta = self.theta  # Start from current estimate         se = 10.0         delta = conv + 1                  th_max = max(self.difficulties) + 0.5         th_min = min(self.difficulties) - 0.5                  if sum(self.responses) == J:             theta = th_max         elif sum(self.responses) == 0:             theta = th_min         else:             while abs(delta) > conv:                 sumnum = 0.0                 sumdem = 0.0                 for j in range(J):                     phat = 1 / (1.0 + math.exp(-1 * (theta - self.difficulties[j])))                     sumnum += self.responses[j] - phat                     sumdem -= phat * (1 - phat)                 delta = sumnum / sumdem                 theta -= delta                 se = 1 / math.sqrt(-sumdem) if sumdem != 0 else 10.0                  return theta, se          def should_stop(self):         """Check if test should stop"""         if len(self.responses) >= self.max_items:             return True         if self.sem < self.sem_threshold and len(self.responses) > 5:             return True         return False          def get_results(self):         """Get final test results"""         if not self.responses:             return None                  standardized = round(((self.theta - 1.4) / 1.5) * 15 + 100)                  return {             'total_questions': len(self.responses),             'correct_answers': sum(self.responses),             'accuracy': sum(self.responses) / len(self.responses) * 100,             'theta': round(self.theta, 2),             'sem': round(self.sem, 2),             'standardized_score': standardized,             'responses': self.responses,             'difficulties': self.difficulties,             'item_codes': self.item_codes         }
